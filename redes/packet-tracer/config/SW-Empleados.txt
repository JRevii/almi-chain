! =====================================================================
! SW-Empleados · Cisco 2960 · switch de la LAN de empleados
! Conexiones: R1 Gi0/0/1 -> Gig0/1 · equipos en Fa0/1 a Fa0/9
! ANTES DE PEGAR: cambia CLAVE_ENABLE y CLAVE_CONSOLA
! =====================================================================
hostname SW-Empleados
enable secret CLAVE_ENABLE
service password-encryption
no ip domain-lookup
banner motd # Acceso restringido. Solo personal autorizado de Almi-Chain. #
!
interface vlan 1
 ip address 10.13.10.2 255.255.255.0
 no shutdown
ip default-gateway 10.13.10.1
!
! Puertos de equipos: modo acceso y como máximo 2 MAC por puerto
interface range FastEthernet0/1 - 9
 switchport mode access
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
!
! Puertos sin usar: apagados (nadie puede enchufarse)
interface range FastEthernet0/10 - 24 , GigabitEthernet0/2
 description Libre - apagado
 shutdown
!
line con 0
 password CLAVE_CONSOLA
 login
 exec-timeout 5 0
! Sin administración remota: solo por consola
line vty 0 15
 transport input none
!
end
write memory
