! =====================================================================
! R1-AlmiChain · Cisco ISR4331 · router principal del restaurante
! Es la configuración del proyecto validado (redes/packet-tracer/almichain.pkt),
! sacada de su "show running-config".
! Dónde: R1-AlmiChain > pestaña CLI > Enter > "enable" > "configure terminal" > pegar
! ANTES DE PEGAR: cambia CLAVE_ENABLE y CLAVE_ADMIN por claves de 10 caracteres
! o más (lo exige "security passwords min-length 10"). En el .pkt ya están
! puestas y cifradas; no se suben en claro a GitHub.
! =====================================================================
hostname R1-AlmiChain
service password-encryption
security passwords min-length 10
enable secret CLAVE_ENABLE
username admin privilege 15 secret CLAVE_ADMIN
no ip domain-lookup
ip domain-name almichain.local
banner motd ^Acceso restringido a personal autorizado^
!
! --- Interfaces ---
interface GigabitEthernet0/0/0
 description WAN - Operador
 ip address dhcp
 ip nat outside
 no shutdown
!
interface GigabitEthernet0/0/1
 description LAN empleados
 ip address 10.13.10.1 255.255.255.0
 ip nat inside
 no shutdown
!
interface GigabitEthernet0/0/2
 description LAN clientes (aislada)
 ip address 10.13.20.1 255.255.255.0
 ip nat inside
 ip access-group CLIENTES-AISLADOS in
 no shutdown
!
! --- DHCP: .1 a .99 y .200 a .254 quedan para IP fijas ---
ip dhcp excluded-address 10.13.10.1 10.13.10.99
ip dhcp excluded-address 10.13.10.200 10.13.10.254
ip dhcp excluded-address 10.13.20.1 10.13.20.99
ip dhcp excluded-address 10.13.20.200 10.13.20.254
!
ip dhcp pool EMPLEADOS
 network 10.13.10.0 255.255.255.0
 default-router 10.13.10.1
 dns-server 203.0.113.11
ip dhcp pool CLIENTES
 network 10.13.20.0 255.255.255.0
 default-router 10.13.20.1
 dns-server 203.0.113.11
!
! --- NAT: las dos LAN salen a internet con la IP de la WAN ---
ip access-list standard NAT-LAN
 permit 10.13.10.0 0.0.0.255
 permit 10.13.20.0 0.0.0.255
ip nat inside source list NAT-LAN interface GigabitEthernet0/0/0 overload
!
! --- Aislamiento de la red de clientes ---
ip access-list extended CLIENTES-AISLADOS
 remark Permitir que los clientes pidan IP por DHCP
 permit udp any any eq bootps
 remark Clientes no pueden llegar a la red de empleados
 deny ip 10.13.20.0 0.0.0.255 10.13.10.0 0.0.0.255
 remark Clientes no pueden administrar el router
 deny tcp any any eq 22
 remark Todo lo demas (Internet) permitido
 permit ip any any
!
! --- Administración: solo SSH v2 y solo desde la red de empleados ---
ip access-list standard SSH-ADMIN
 permit 10.13.10.0 0.0.0.255
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
!
line con 0
 exec-timeout 5 0
 login local
line vty 0 4
 access-class SSH-ADMIN in
 exec-timeout 5 0
 login local
 transport input ssh
!
end
write memory
