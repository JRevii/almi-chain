! =====================================================================
! R-Operador · Cisco 1941 · simula el router de la fibra contratada
! Da IP por DHCP a la WAN del local y conecta con "internet" (203.0.113.0/24,
! rango reservado para documentación)
! ANTES DE PEGAR: cambia CLAVE_ENABLE
! =====================================================================
hostname R-Operador
enable secret CLAVE_ENABLE
service password-encryption
no ip domain-lookup
!
interface GigabitEthernet0/0
 description Hacia el local (WAN de R1-AlmiChain)
 ip address 192.168.1.1 255.255.255.0
 no shutdown
!
interface GigabitEthernet0/1
 description Internet (simulado)
 ip address 203.0.113.1 255.255.255.0
 no shutdown
!
ip dhcp excluded-address 192.168.1.1 192.168.1.9
ip dhcp pool WAN-CLIENTE
 network 192.168.1.0 255.255.255.0
 default-router 192.168.1.1
 dns-server 1.1.1.1
!
end
write memory
